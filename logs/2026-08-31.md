  - 연/월 필터링을 `TO_CHAR(meal_date, 'YYYY'/'MM')` 문자열 비교로 구현한 핵심 조회 쿼리:
    ```xml
    <select id="selectCalendarItems" parameterType="map" resultType="com.sist.web.vo.CalendarItemVO">
      SELECT ci.id AS id, ci.user_id as user_id,
             ci.meal_date as meal_date, ci.meal_type as meal_type,
             ci.RCP_SEQ as rcp_seq, r.RCP_NM as rcp_nm,
             r.ATT_FILE_NO_MAIN as att_file_no_main
      FROM CALENDAR_ITEM ci
      JOIN RECIPE r ON ci.RCP_SEQ = r.RCP_SEQ
      WHERE ci.user_id = #{user_id}
        AND TO_CHAR(ci.meal_date, 'YYYY') = #{year}
        AND TO_CHAR(ci.meal_date, 'MM') = #{month}
      ORDER BY ci.meal_date, ci.meal_type
    </select>
    ```
    (파일: `src/main/resources/mybatis/mapper/calendarItem-mapper.xml`, 커밋: `cc47851`)

    ```js
    const calendarCells = computed(() => {
      const firstDay = new Date(year.value, month.value - 1, 1)
      const lastDay = new Date(year.value, month.value, 0)
      const startWeekday = firstDay.getDay()
      const totalDays = lastDay.getDate()

      const cells = []
      for (let i = 0; i < startWeekday; i++) cells.push({ empty: true })

      for (let d = 1; d <= totalDays; d++) {
        const dateStr = `${year.value}-${String(month.value).padStart(2, "0")}-${String(d).padStart(2, "0")}`
        cells.push({
          empty: false,
          day: d,
          dateStr,
          meals: MEAL_TYPES.map((type) => ({
            type,
            data: itemMap.value[`${dateStr}_${type}`] || null,
          })),
        })
      }
      return cells
    })
    ```
    (파일: `src/composables/useCalendarItems.js`, 커밋: `bc4f2ee`)
